//====================================================
// FUNCTIONS

truncate(number, decimals) =>
    factor = math.pow(10, decimals)
    int(number * factor) / factor

f_tfInMinutes() =>
    _tf = timeframe.period == "1" ? "3" :
         timeframe.period == "3" ? "5" :
         timeframe.period == "5" ? "15" :
         timeframe.period == "15" ? "30" :
         timeframe.period == "30" ? "60" :
         timeframe.period == "60" ? "240" : "D"
    _tf

f_resInMinutes() =>
    timeframe.multiplier * (
         timeframe.isseconds ? 1.0 / 60.0 :
         timeframe.isminutes ? 1.0 :
         timeframe.isdaily ? 1440.0 :
         timeframe.isweekly ? 10080.0 :
         timeframe.ismonthly ? 43800.0 : na)

f_deleteLine(_line) =>
    if not na(_line)
        line.delete(_line)

     true,

     group="SIGNAL SETTINGS")

//====================================================
// ORIGINAL RSI FILTER
//====================================================

rsiLength = input.int(
     7,
     "RSI Length",
     minval=1,
     group="ORIGINAL FILTER")

     45,
     "RSI Top",
     group="ORIGINAL FILTER")

bottomRSI = input.int(
     10,
     "RSI Bottom",
     group="ORIGINAL FILTER")

//====================================================

filter1 = "Filter with Atr"
filter2 = "Filter with RSI"
filter3 = "Atr or RSI"
filter4 = "Atr and RSI"
filter5 = "No Filtering"
filter6 = "Entry Only in sideways market(By ATR or RSI)"
filter7 = "Entry Only in sideways market(By ATR and RSI)"

typeFilter = input.string(
     filter5,
     "Sideways Filter",
     options=[
         filter1,
         filter2,
         filter3,
         filter4,
         filter5,
         filter6,
         filter7
     ],
     group="ORIGINAL FILTER")

originalATRLength = 5
originalMALength = 5

originalATR = request.security(
     syminfo.tickerid,
     "",
     ta.atr(originalATRLength))

originalATRMA = ta.ema(
     originalATR,
     originalMALength)

atrCondition =
     originalATR >= originalATRMA

rsiCondition =
     RSI > topRSI or RSI < bottomRSI

atrOrRSI =
     atrCondition or rsiCondition

atrAndRSI =
     atrCondition and rsiCondition

sidewaysATR =
     originalATR <= originalATRMA

sidewaysRSI =
     RSI < topRSI and RSI > bottomRSI

sidewaysOR =
     sidewaysATR or sidewaysRSI

sidewaysAND =
     sidewaysATR and sidewaysRSI

originalFilter =
     typeFilter == filter1 ? atrCondition :
     typeFilter == filter2 ? rsiCondition :
     typeFilter == filter3 ? atrOrRSI :
     typeFilter == filter4 ? atrAndRSI :
     typeFilter == filter5 ? true :
     typeFilter == filter6 ? sidewaysOR :
     typeFilter == filter7 ? sidewaysAND :
     true

originalFilterOK =
     not na(originalFilter) and originalFilter

resolutionMinutes = f_resInMinutes()

     math.max(
         1,
         math.round(resolutionMinutes * tfmult))

// timeframe string.
//
// Examples:
// 54     -> "54"
// 240    -> "240"
// 1440   -> "1D"
// 25920  -> "18D"

myTime =
     calculatedMinutes >= 1440 ?
     str.tostring(
         math.round(calculatedMinutes / 1440)) + "D" :
     str.tostring(calculatedMinutes)

     close < ema200

//----------------------------------------------------

//----------------------------------------------------

useADX = input.bool(
     true,
     "ADX Strength",
     group=smartGroup)

adxLength = input.int(
     14,
     "ADX Length",
     minval=1,
     group=smartGroup)

     14,
     "ADX Smoothing",
     minval=1,
     group=smartGroup)

     20.0,
     "Minimum ADX",
     minval=1,
     step=0.5,
     group=smartGroup)

[diPlus, diMinus, adx] =
     ta.dmi(
         adxLength,
         adxSmoothing)

adxBuy =
     adx >= minimumADX and
     diPlus > diMinus

adxSell =
     adx >= minimumADX and
     diMinus > diPlus

//----------------------------------------------------
// VWAP
//----------------------------------------------------

useVWAPDirection = input.bool(
     true,
     "VWAP Direction",
     group=smartGroup)

vwap =
     ta.vwap(hlc3)

vwapBuy =
     close > vwap

vwapSell =
     close < vwap

//----------------------------------------------------
// ENTRY LOCATION
//----------------------------------------------------

useLocationFilter = input.bool(
     true,
     "Don't Chase Price",
     group=smartGroup)

locationATR =
     ta.atr(14)

vwapDistance =
     locationATR > 0 ?
     math.abs(close - vwap) / locationATR :
     999.0

maxVWAPDistance = input.float(
     1.5,
     "Max Distance From VWAP (ATR)",
     minval=0.1,
     step=0.1,
     group=smartGroup)

locationOK =
     vwapDistance <= maxVWAPDistance

//====================================================
// FINAL SMART FILTER
//====================================================

buySmartFilter =
     (not useEMA200 or trendBuy) and
     (not useADX or adxBuy) and
     (not useVWAPDirection or vwapBuy) and
     (not useLocationFilter or locationOK)

sellSmartFilter =
     (not useEMA200 or trendSell) and
     (not useADX or adxSell) and
     (not useVWAPDirection or vwapSell) and
     (not useLocationFilter or locationOK)

//====================================================
// STRUCTURE RISK
//====================================================

riskGroup = "STRUCTURE RISK"

useStructureSL = input.bool(
     true,
     "Use Swing Structure SL",
     group=riskGroup)

swingLookback = input.int(
     5,
     "Swing Lookback",
     minval=2,
     group=riskGroup)

structureBufferATR = input.float(
     0.20,
     "Structure Buffer ATR",
     minval=0,
     step=0.05,
     group=riskGroup)

atrStopMultiplier = input.float(
     1.0,
     "ATR Stop Multiplier",
     minval=0.1,
     step=0.1,
     group=riskGroup)

maxRiskATR = input.float(
     2.0,
     "Maximum Risk (ATR)",
     minval=0.5,
     step=0.1,
     group=riskGroup)

tradeATR =
     ta.atr(14)

atrRisk =
     tradeATR * atrStopMultiplier

previousSwingLow =
     ta.lowest(
         low[1],
         swingLookback)

previousSwingHigh =
     ta.highest(
         high[1],
         swingLookback)

structureBuySL =
     previousSwingLow -
     tradeATR * structureBufferATR

structureSellSL =
     previousSwingHigh +
     tradeATR * structureBufferATR

proposedBuySL =
     useStructureSL ?
     structureBuySL :
     close - atrRisk

proposedSellSL =
     useStructureSL ?
     structureSellSL :
     close + atrRisk

buyRisk =
     close - proposedBuySL

sellRisk =
     proposedSellSL - close

buyRiskATR =
     tradeATR > 0 ?
     buyRisk / tradeATR :
     999.0

sellRiskATR =
     tradeATR > 0 ?
     sellRisk / tradeATR :
     999.0

buyRiskOK =
     buyRisk > 0 and
     buyRiskATR <= maxRiskATR

sellRiskOK =
     sellRisk > 0 and
     sellRiskATR <= maxRiskATR

//====================================================
// SMART REFRESH / RE-ENTRY
//====================================================

refreshGroup = "SMART REFRESH"

useRefreshEntries = input.bool(
     true,
     "Allow Refresh Entries",
     group=refreshGroup)

refreshBars = input.int(
     12,
     "Refresh Every N Bars",
     minval=1,
     group=refreshGroup)

var int lastExitBar = na

barsSinceExit =
     na(lastExitBar) ?
     999999 :
     bar_index - lastExitBar

refreshReady =
     not na(lastExitBar) and
     barsSinceExit >= refreshBars

refreshBuy =
     useRefreshEntries and
     refreshReady and
     buySmartFilter and
     buyRiskOK and
     dateAllowed

refreshSell =
     useRefreshEntries and
     refreshReady and
     sellSmartFilter and
     sellRiskOK and
     dateAllowed

//====================================================
// FINAL SIGNAL
//====================================================

buySignal =
     (
         originalBuy and
         buySmartFilter and
         buyRiskOK and
         dateAllowed
     ) or refreshBuy

sellSignal =
     (
         originalSell and
         sellSmartFilter and
         sellRiskOK and
         dateAllowed
     ) or refreshSell

tradeGroup = "TRADE MANAGEMENT"

showLevels = input.bool(

     "Show Current Trade Levels",

showEntryLabels = input.bool(

showResultLabels = input.bool(

//====================================================
// TRADE VARIABLES
//====================================================

var bool inTrade = false

var int direction = 0

var float entry = na
var float originalSL = na
var float activeSL = na

var float tp1 = na
var float tp2 = na
var float tp3 = na

var float risk = na

// 0 = no target
// 1 = TP1
// 2 = TP2
// 3 = TP3

var int highestTarget = 0

var bool protectionActive = false

//====================================================
// CURRENT LINES
//====================================================

var line entryLine = na
var line slLine = na
var line tp1Line = na
var line tp2Line = na
var line tp3Line = na

//====================================================
// STATISTICS
//====================================================

var int totalTrades = 0
var int completedTrades = 0

var int slCount = 0
var int tp1Count = 0
var int tp2Count = 0
var int tp3Count = 0

var float totalR = 0.0

//====================================================
// EVENTS
//====================================================

bool buyEvent = false
bool sellEvent = false

bool tp1Event = false
bool tp2Event = false
bool tp3Event = false

bool slEvent = false
bool beEvent = false

//====================================================
// BUY ENTRY
//====================================================

if not inTrade and buySignal

    f_deleteLine(entryLine)
    f_deleteLine(slLine)
    f_deleteLine(tp1Line)
    f_deleteLine(tp2Line)
    f_deleteLine(tp3Line)

    inTrade := true
    direction := 1

    entry := close
    risk := buyRisk

    originalSL := proposedBuySL
    activeSL := originalSL

    tp1 := entry + risk
    tp2 := entry + risk * 2.0
    tp3 := entry + risk * 3.0

    highestTarget := 0
    protectionActive := false

    totalTrades += 1
    buyEvent := true

    if showLevels

        entryLine := line.new(
             bar_index,
             entry,
             bar_index + 1,
             entry,
             extend=extend.right,
             color=color.blue,
             width=2)

        slLine := line.new(
             bar_index,
             originalSL,
             bar_index + 1,
             originalSL,
             extend=extend.right,
             color=color.red,
             width=2)

        tp1Line := line.new(
             bar_index,
             tp1,
             bar_index + 1,
             tp1,
             extend=extend.right,
             color=color.green,
             style=line.style_dashed)

        tp2Line := line.new(
             bar_index,
             tp2,
             bar_index + 1,
             tp2,
             extend=extend.right,
             color=color.green,
             style=line.style_dashed)

        tp3Line := line.new(
             bar_index,
             tp3,
             bar_index + 1,
             tp3,
             extend=extend.right,
             color=color.green,
             style=line.style_dashed)

    if showEntryLabels

        label.new(
             bar_index,
             low,
             "BUY",
             style=label.style_label_up,
             color=color.green,
             textcolor=color.white,
             size=size.small)

//====================================================
// SELL ENTRY
//====================================================

if not inTrade and sellSignal

    f_deleteLine(entryLine)
    f_deleteLine(slLine)
    f_deleteLine(tp1Line)
    f_deleteLine(tp2Line)
    f_deleteLine(tp3Line)

    inTrade := true
    direction := -1

    entry := close
    risk := sellRisk

    originalSL := proposedSellSL
    activeSL := originalSL

    tp1 := entry - risk
    tp2 := entry - risk * 2.0
    tp3 := entry - risk * 3.0

    highestTarget := 0
    protectionActive := false

    totalTrades += 1
    sellEvent := true

    if showLevels

        entryLine := line.new(
             bar_index,
             entry,
             bar_index + 1,
             entry,
             extend=extend.right,
             color=color.blue,
             width=2)

        slLine := line.new(
             bar_index,
             originalSL,
             bar_index + 1,
             originalSL,
             extend=extend.right,
             color=color.red,
             width=2)

        tp1Line := line.new(
             bar_index,
             tp1,
             bar_index + 1,
             tp1,
             extend=extend.right,
             color=color.green,
             style=line.style_dashed)

        tp2Line := line.new(
             bar_index,
             tp2,
             bar_index + 1,
             tp2,
             extend=extend.right,
             color=color.green,
             style=line.style_dashed)

        tp3Line := line.new(
             bar_index,
             tp3,
             bar_index + 1,
             tp3,
             extend=extend.right,
             color=color.green,
             style=line.style_dashed)

    if showEntryLabels

        label.new(
             bar_index,
             high,
             "SELL",
             style=label.style_label_down,
             color=color.red,
             textcolor=color.white,
             size=size.small)

//====================================================
// BUY MANAGEMENT
//====================================================

if inTrade and direction == 1

    hitSL = low <= activeSL
    hitTP1 = high >= tp1
    hitTP2 = high >= tp2
    hitTP3 = high >= tp3

    // BEFORE TP1
    if highestTarget == 0

        if hitTP3

            highestTarget := 3
            completedTrades += 1
            tp3Count += 1
            totalR += 3.0
            tp3Event := true

            if showResultLabels
                label.new(
                     bar_index,
                     high,
                     "TP3 ✅  +3R",
                     style=label.style_label_down,
                     color=color.green,
                     textcolor=color.white)

            f_deleteLine(entryLine)
            f_deleteLine(slLine)
            f_deleteLine(tp1Line)
            f_deleteLine(tp2Line)
            f_deleteLine(tp3Line)

            inTrade := false
            direction := 0

        else if hitTP2

            highestTarget := 2
            protectionActive := true
            activeSL := entry
            tp2Event := true

            if not na(slLine)
                line.set_y1(slLine, entry)
                line.set_y2(slLine, entry)
                line.set_color(slLine, color.orange)

            if showResultLabels
                label.new(
                     bar_index,
                     high,
                     "TP2 ✅\nBE 🛡️",
                     style=label.style_label_down,
                     color=color.green,
                     textcolor=color.white)

        else if hitTP1

            highestTarget := 1
            protectionActive := true
            activeSL := entry
            tp1Event := true

            if not na(slLine)
                line.set_y1(slLine, entry)
                line.set_y2(slLine, entry)
                line.set_color(slLine, color.orange)

            if showResultLabels
                label.new(
                     bar_index,
                     high,
                     "TP1 ✅\nBE 🛡️",
                     style=label.style_label_down,
                     color=color.green,
                     textcolor=color.white)

        else if hitSL

            completedTrades += 1
            slCount += 1
            totalR -= 1.0
            slEvent := true

            if showResultLabels
                label.new(
                     bar_index,
                     low,
                     "SL ❌  -1R",
                     style=label.style_label_up,
                     color=color.red,
                     textcolor=color.white)

            f_deleteLine(entryLine)
            f_deleteLine(slLine)
            f_deleteLine(tp1Line)
            f_deleteLine(tp2Line)
            f_deleteLine(tp3Line)

            inTrade := false
            direction := 0

    // AFTER TP1 / TP2
    else

        if hitTP3

            highestTarget := 3
            completedTrades += 1
            tp3Count += 1
            totalR += 3.0
            tp3Event := true

            if showResultLabels
                label.new(
                     bar_index,
                     high,
                     "TP3 ✅  +3R",
                     style=label.style_label_down,
                     color=color.green,
                     textcolor=color.white)

            f_deleteLine(entryLine)
            f_deleteLine(slLine)
            f_deleteLine(tp1Line)
            f_deleteLine(tp2Line)
            f_deleteLine(tp3Line)

            inTrade := false
            direction := 0

        else if hitTP2 and highestTarget < 2

            highestTarget := 2
            protectionActive := true
            activeSL := entry
            tp2Event := true

            if not na(slLine)
                line.set_y1(slLine, entry)
                line.set_y2(slLine, entry)
                line.set_color(slLine, color.orange)

            if showResultLabels
                label.new(
                     bar_index,
                     high,
                     "TP2 ✅\nBE 🛡️",
                     style=label.style_label_down,
                     color=color.green,
                     textcolor=color.white)

        else if protectionActive and low <= entry

            completedTrades += 1

            if highestTarget == 1
                tp1Count += 1
                totalR += 1.0
            else
                tp2Count += 1
                totalR += 2.0

            beEvent := true

            if showResultLabels
                label.new(
                     bar_index,
                     entry,
                     "BE 🛡️\n+" +
                     str.tostring(highestTarget) +
                     "R",
                     style=label.style_label_left,
                     color=color.orange,
                     textcolor=color.white)

            f_deleteLine(entryLine)
            f_deleteLine(slLine)
            f_deleteLine(tp1Line)
            f_deleteLine(tp2Line)
            f_deleteLine(tp3Line)

            inTrade := false
            direction := 0

//====================================================
// SELL MANAGEMENT
//====================================================

if inTrade and direction == -1

    hitSL = high >= activeSL
    hitTP1 = low <= tp1
    hitTP2 = low <= tp2
    hitTP3 = low <= tp3

    // BEFORE TP1
    if highestTarget == 0

        if hitTP3

            highestTarget := 3
            completedTrades += 1
            tp3Count += 1
            totalR += 3.0
            tp3Event := true

            if showResultLabels
                label.new(
                     bar_index,
                     low,
                     "TP3 ✅  +3R",
                     style=label.style_label_up,
                     color=color.green,
                     textcolor=color.white)

            f_deleteLine(entryLine)
            f_deleteLine(slLine)
            f_deleteLine(tp1Line)
            f_deleteLine(tp2Line)
            f_deleteLine(tp3Line)

            inTrade := false
            direction := 0

        else if hitTP2

            highestTarget := 2
            protectionActive := true
            activeSL := entry
            tp2Event := true

            if not na(slLine)
                line.set_y1(slLine, entry)
                line.set_y2(slLine, entry)
                line.set_color(slLine, color.orange)

            if showResultLabels
                label.new(
                     bar_index,
                     low,
                     "TP2 ✅\nBE 🛡️",
                     style=label.style_label_up,
                     color=color.green,
                     textcolor=color.white)

        else if hitTP1

            highestTarget := 1
            protectionActive := true
            activeSL := entry
            tp1Event := true

            if not na(slLine)
                line.set_y1(slLine, entry)
                line.set_y2(slLine, entry)
                line.set_color(slLine, color.orange)

            if showResultLabels
                label.new(
                     bar_index,
                     low,
                     "TP1 ✅\nBE 🛡️",
                     style=label.style_label_up,
                     color=color.green,
                     textcolor=color.white)

        else if hitSL

            completedTrades += 1
            slCount += 1
            totalR -= 1.0
            slEvent := true

            if showResultLabels
                label.new(
                     bar_index,
                     high,
                     "SL ❌  -1R",
                     style=label.style_label_down,
                     color=color.red,
                     textcolor=color.white)

            f_deleteLine(entryLine)
            f_deleteLine(slLine)
            f_deleteLine(tp1Line)
            f_deleteLine(tp2Line)
            f_deleteLine(tp3Line)

            inTrade := false
            direction := 0

    // AFTER TP1 / TP2
    else

        if hitTP3

            highestTarget := 3
            completedTrades += 1
            tp3Count += 1
            totalR += 3.0
            tp3Event := true

            if showResultLabels
                label.new(
                     bar_index,
                     low,
                     "TP3 ✅  +3R",
                     style=label.style_label_up,
                     color=color.green,
                     textcolor=color.white)

            f_deleteLine(entryLine)
            f_deleteLine(slLine)
            f_deleteLine(tp1Line)
            f_deleteLine(tp2Line)
            f_deleteLine(tp3Line)

            inTrade := false
            direction := 0

        else if hitTP2 and highestTarget < 2

            highestTarget := 2
            protectionActive := true
            activeSL := entry
            tp2Event := true

            if not na(slLine)
                line.set_y1(slLine, entry)
                line.set_y2(slLine, entry)
                line.set_color(slLine, color.orange)

            if showResultLabels
                label.new(
                     bar_index,
                     low,
                     "TP2 ✅\nBE 🛡️",
                     style=label.style_label_up,
                     color=color.green,
                     textcolor=color.white)

        else if protectionActive and high >= entry

            completedTrades += 1

            if highestTarget == 1
                tp1Count += 1
                totalR += 1.0
            else
                tp2Count += 1
                totalR += 2.0

            beEvent := true

            if showResultLabels
                label.new(
                     bar_index,
                     entry,
                     "BE 🛡️\n+" +
                     str.tostring(highestTarget) +
                     "R",
                     style=label.style_label_right,
                     color=color.orange,
                     textcolor=color.white)

            f_deleteLine(entryLine)
            f_deleteLine(slLine)
            f_deleteLine(tp1Line)
            f_deleteLine(tp2Line)
            f_deleteLine(tp3Line)

            inTrade := false
            direction := 0
